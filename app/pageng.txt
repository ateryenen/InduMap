type StrategyLevel = "L1" | "L2" | "L3" | "Info";

interface BaseNode {
  id: string;
  type: NodeType;
  x: number;
  y: number;
  width: number;
  height: number;
  text?: string;
  bgColor?: string;
  fontSize?: number;
  owner?: string;
  level?: StrategyLevel;
  categoryTag?: string;
  impact?: number;

}

interface TableNode extends BaseNode {
  type: "table";
  rows: number;
  cols: number;
  cells: string[][];
}

interface ImageNode extends BaseNode {
  type: "image";
  imageDataUrl: string;
}

type Node = BaseNode | TableNode | ImageNode;

interface Edge {
  id: string;
  fromId: string;
  toId: string;
}

interface Viewport {
  x: number;
  y: number;
  scale: number;
}

interface BoardState {
  nodes: Node[];
  edges: Edge[];
  viewport: Viewport;
  projectName?: string;
}

/** 多頁專案 */
interface Page {
  id: string;
  name: string;
  nodes: Node[];
  edges: Edge[];
  viewport: Viewport;
}

interface ProjectFileV2 {
  version: 2;
  projectName: string;
  activePageId: string | null;
  pages: Page[];
}

type ResizeHandle =
  | "top-left"
  | "top"
  | "top-right"
  | "right"
  | "bottom-right"
  | "bottom"
  | "bottom-left"
  | "left"
  | null;

type SelectionMode = "replace" | "add" | "subtract";

/** 啟動時從 localStorage 載入 */
function loadInitialState(): {
  projectName: string;
  pages: Page[];
  activePageId: string | null;
} {
  if (typeof window === "undefined") {
    return {
      projectName: "未命名專案",
      pages: [defaultPage],
      activePageId: defaultPage.id,
    };
  }
  try {
    const raw = window.localStorage.getItem(STORAGE_KEY);
    if (!raw) {
      return {
        projectName: "未命名專案",
        pages: [defaultPage],
        activePageId: defaultPage.id,
      };
    }
    const parsed = JSON.parse(raw) as ProjectFileV2;
    if (parsed.version === 2 && parsed.pages?.length) {
      return {
        projectName: parsed.projectName || "未命名專案",
        pages: parsed.pages,
        activePageId: parsed.activePageId || parsed.pages[0].id,
      };
    }
  } catch (err) {
    console.error("Failed to load from localStorage", err);
  }
  return {
    projectName: "未命名專案",
    pages: [defaultPage],
    activePageId: defaultPage.id,
  };
}

interface DragState {
  offsetX: number;
  offsetY: number;
  nodeId: string;
  isDragging: boolean;
}

interface ResizeState {
  nodeId: string | null;
  handle: ResizeHandle;
  startX: number;
  startY: number;
  startWidth: number;
  startHeight: number;
}

interface PanState {
  isPanning: boolean;
  startX: number;
  startY: number;
  startViewportX: number;
  startViewportY: number;
}

interface CanvasSelectionState {
  isSelecting: boolean;
  startX: number;
  startY: number;
}

export default function WhiteboardApp() {
  const {
    projectName: initialProjectName,
    pages: initialPages,
    activePageId: initialActivePageId,
  } = loadInitialState();

  const [projectName, setProjectName] = useState(initialProjectName);
  const [pages, setPages] = useState<Page[]>(initialPages);
  const [activePageId, setActivePageId] = useState<string | null>(
    initialActivePageId
  );

  const [tool, setTool] = useState<Tool>("select");
  const [viewport, setViewport] = useState<Viewport>(() => {
    const page = initialPages.find((p) => p.id === initialActivePageId);
    return page?.viewport || defaultViewport;
  });
  const [nodes, setNodes] = useState<Node[]>(() => {
    const page = initialPages.find((p) => p.id === initialActivePageId);
    return page?.nodes || [];
  });
  const [edges, setEdges] = useState<Edge[]>(() => {
    const page = initialPages.find((p) => p.id === initialActivePageId);
    return page?.edges || [];
  });

  const [selectedIds, setSelectedIds] = useState<string[]>([]);
  const [primaryId, setPrimaryId] = useState<string | null>(null);
  const [editingId, setEditingId] = useState<string | null>(null);

  const [dragState, setDragState] = useState<DragState | null>(null);
  const [resizeState, setResizeState] = useState<ResizeState>({
    nodeId: null,
    handle: null,
    startX: 0,
    startY: 0,
    startWidth: 0,
    startHeight: 0,
  });
  const [panState, setPanState] = useState<PanState>({
    isPanning: false,
    startX: 0,
    startY: 0,
    startViewportX: 0,
    startViewportY: 0,
  });
  const [canvasSelection, setCanvasSelection] =
    useState<CanvasSelectionState | null>(null);
  const [selectionRect, setSelectionRect] = useState<{
    x: number;
    y: number;
    w: number;
    h: number;
  } | null>(null);

  const [connectingFrom, setConnectingFrom] = useState<string | null>(null);

  const [formatPainterActive, setFormatPainterActive] = useState(false);
  const [copiedStyle, setCopiedStyle] = useState<
    | {
        bgColor?: string;
        fontSize?: number;
        owner?: string;
        level?: StrategyLevel;
        categoryTag?: string;
      }
    | null
  >(null);

  const [clipboardNodes, setClipboardNodes] = useState<Node[] | null>(null);
  const [clipboardEdges, setClipboardEdges] = useState<Edge[] | null>(null);

  const [history, setHistory] = useState<BoardState[]>([]);
  const [future, setFuture] = useState<BoardState[]>([]);

  const [isSidebarOpen, setIsSidebarOpen] = useState(true);
  const [isOutlineOpen, setIsOutlineOpen] = useState(true);
  const [isRightPanelOpen, setIsRightPanelOpen] = useState(true);

  const containerRef = useRef<HTMLDivElement | null>(null);
  const canvasRef = useRef<HTMLDivElement | null>(null);

  const isDraggingRef = useRef(false);
  const isResizingRef = useRef(false);
  const isPanningRef = useRef(false);

  const [selectionMode, setSelectionMode] =
    useState<SelectionMode>("replace");
  const selectionModeRef = useRef<SelectionMode>("replace");

  useEffect(() => {
    selectionModeRef.current = selectionMode;
  }, [selectionMode]);

  /** 每次狀態變更就存回 localStorage（包含多頁） */
  useEffect(() => {
    if (typeof window === "undefined") return;
    const data: ProjectFileV2 = {
      version: 2,
      projectName,
      activePageId,
      pages: pages.map((p) =>
        p.id === activePageId
          ? { ...p, nodes, edges, viewport }
          : { ...p }
      ),
    };
    window.localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
  }, [pages, projectName, activePageId, nodes, edges, viewport]);

  const recordHistory = () => {
    setHistory((prev) => [
      ...prev,
      {
        nodes: JSON.parse(JSON.stringify(nodes)),
        edges: JSON.parse(JSON.stringify(edges)),
        viewport: { ...viewport },
        projectName,
      },
    ]);
    setFuture([]);
  };

  const undo = () => {
    setHistory((prev) => {
      if (!prev.length) return prev;
      const last = prev[prev.length - 1];
      setFuture((f) => [
        {
          nodes: JSON.parse(JSON.stringify(nodes)),
          edges: JSON.parse(JSON.stringify(edges)),
          viewport: { ...viewport },
          projectName,
        },
        ...f,
      ]);
      setNodes(last.nodes);
      setEdges(last.edges);
      setViewport(last.viewport);
      setProjectName(last.projectName ?? projectName);
      return prev.slice(0, -1);
    });
  };

  const redo = () => {
    setFuture((prev) => {
      if (!prev.length) return prev;
      const next = prev[0];
      setHistory((h) => [
        ...h,
        {
          nodes: JSON.parse(JSON.stringify(nodes)),
          edges: JSON.parse(JSON.stringify(edges)),
          viewport: { ...viewport },
          projectName,
        },
      ]);
      setNodes(next.nodes);
      setEdges(next.edges);
      setViewport(next.viewport);
      setProjectName(next.projectName ?? projectName);
      return prev.slice(1);
    });
  };

  const handleNodeMouseDown = (
    e: MouseEvent<HTMLDivElement>,
    node: Node
  ) => {
    e.stopPropagation();
    const rect = containerRef.current?.getBoundingClientRect();
    if (!rect) return;
    const worldX = (e.clientX - rect.left - viewport.x) / viewport.scale;
    const worldY = (e.clientY - rect.top - viewport.y) / viewport.scale;

    const offsetX = worldX - node.x;
    const offsetY = worldY - node.y;

    const isMultiSelect =
      e.shiftKey || e.metaKey || e.ctrlKey || selectionMode !== "replace";

    let newSelectedIds = selectedIds;
    if (isMultiSelect) {
      if (!selectedIds.includes(node.id)) {
        newSelectedIds = [...selectedIds, node.id];
      } else if (selectionMode === "subtract") {
        newSelectedIds = selectedIds.filter((id) => id !== node.id);
      }
    } else {
      if (!selectedIds.includes(node.id)) {
        newSelectedIds = [node.id];
      }
    }
    setSelectedIds(newSelectedIds);
    setPrimaryId(node.id);
    setEditingId(null);

    let pivotNode = node;
    if (newSelectedIds.length > 1) {
      const currentIndex = newSelectedIds.indexOf(node.id);
      const nextIndex =
        (currentIndex + 1 + newSelectedIds.length) % newSelectedIds.length;
      const nextId = newSelectedIds[nextIndex];
      const maybeNode = nodes.find((n) => n.id === nextId);
      if (maybeNode) {
        pivotNode = maybeNode;
      }
    }

    const pivotOffsetX = worldX - pivotNode.x;
    const pivotOffsetY = worldY - pivotNode.y;

    setDragState({
      offsetX: pivotOffsetX,
      offsetY: pivotOffsetY,
      nodeId: pivotNode.id,
      isDragging: true,
    });
    isDraggingRef.current = true;
  };

  const handleCanvasMouseDown = (e: MouseEvent<HTMLDivElement>) => {
    if (!containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    const worldX = (e.clientX - rect.left - viewport.x) / viewport.scale;
    const worldY = (e.clientY - rect.top - viewport.y) / viewport.scale;

    if (tool === "pan" || e.button === 1) {
      setPanState({
        isPanning: true,
        startX: e.clientX,
        startY: e.clientY,
        startViewportX: viewport.x,
        startViewportY: viewport.y,
      });
      isPanningRef.current = true;
      return;
    }

    recordHistory();
    setCanvasSelection({
      isSelecting: true,
      startX: worldX,
      startY: worldY,
    });
    setSelectionRect({
      x: worldX,
      y: worldY,
      w: 0,
      h: 0,
    });

    if (selectionMode === "replace") {
      setSelectedIds([]);
      setPrimaryId(null);
      setEditingId(null);
    }
  };

  const handleCanvasMouseMove = (e: MouseEvent<HTMLDivElement>) => {
    if (!containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    const worldX = (e.clientX - rect.left - viewport.x) / viewport.scale;
    const worldY = (e.clientY - rect.top - viewport.y) / viewport.scale;

    if (isDraggingRef.current && dragState?.isDragging) {
      const pivot = nodes.find((n) => n.id === dragState.nodeId);
      if (!pivot) return;
      const dxPivot = worldX - dragState.offsetX - pivot.x;
      const dyPivot = worldY - dragState.offsetY - pivot.y;

      setNodes((prev) =>
        prev.map((node) =>
          selectedIds.includes(node.id)
            ? { ...node, x: node.x + dxPivot, y: node.y + dyPivot }
            : node
        )
      );
      return;
    }

    if (isResizingRef.current && resizeState.nodeId && resizeState.handle) {
      const dx =
        (e.clientX - rect.left) / viewport.scale - resizeState.startX;
      const dy =
        (e.clientY - rect.top) / viewport.scale - resizeState.startY;

      setNodes((prev) =>
        prev.map((node) => {
          if (node.id !== resizeState.nodeId) return node;

          let { x, y, width, height } = node;
          let newX = x;
          let newY = y;
          let newWidth = width;
          let newHeight = height;
          const minSize = 40;

          if (
            resizeState.handle === "top-left" ||
            resizeState.handle === "top" ||
            resizeState.handle === "top-right"
          ) {
            newHeight = Math.max(minSize, resizeState.startHeight - dy);
            newY = y + (height - newHeight);
          }
          if (
            resizeState.handle === "bottom-left" ||
            resizeState.handle === "bottom" ||
            resizeState.handle === "bottom-right"
          ) {
            newHeight = Math.max(minSize, resizeState.startHeight + dy);
          }
          if (
            resizeState.handle === "top-left" ||
            resizeState.handle === "left" ||
            resizeState.handle === "bottom-left"
          ) {
            newWidth = Math.max(minSize, resizeState.startWidth - dx);
            newX = x + (width - newWidth);
          }
          if (
            resizeState.handle === "top-right" ||
            resizeState.handle === "right" ||
            resizeState.handle === "bottom-right"
          ) {
            newWidth = Math.max(minSize, resizeState.startWidth + dx);
          }

          return {
            ...node,
            x: newX,
            y: newY,
            width: newWidth,
            height: newHeight,
          };
        })
      );
      return;
    }

    if (isPanningRef.current && panState.isPanning) {
      const dx = e.clientX - panState.startX;
      const dy = e.clientY - panState.startY;
      setViewport((v) => ({
        ...v,
        x: panState.startViewportX + dx,
        y: panState.startViewportY + dy,
      }));
      return;
    }

    if (canvasSelection?.isSelecting && selectionRect) {
      const sx = Math.min(canvasSelection.startX, worldX);
      const sy = Math.min(canvasSelection.startY, worldY);
      const sw = Math.abs(worldX - canvasSelection.startX);
      const sh = Math.abs(worldY - canvasSelection.startY);

      setSelectionRect({
        x: sx,
        y: sy,
        w: sw,
        h: sh,
      });
    }
  };

  const handleCanvasMouseUp = () => {
    if (canvasSelection?.isSelecting && selectionRect) {
      const { x, y, w, h } = selectionRect;
      const includedIds = nodes
        .filter(
          (node) =>
            node.x + node.width >= x &&
            node.x <= x + w &&
            node.y + node.height >= y &&
            node.y <= y + h
        )
        .map((node) => node.id);

      let newSelected: string[] = [];
      if (selectionModeRef.current === "add") {
        const set = new Set([...selectedIds, ...includedIds]);
        newSelected = Array.from(set);
      } else if (selectionModeRef.current === "subtract") {
        const set = new Set(selectedIds);
        includedIds.forEach((id) => set.delete(id));
        newSelected = Array.from(set);
      } else {
        newSelected = includedIds;
      }

      setSelectedIds(newSelected);
      setPrimaryId(newSelected[0] || null);
      setSelectionRect(null);
    }

    isDraggingRef.current = false;
    isResizingRef.current = false;
    isPanningRef.current = false;
    setDragState(null);
    setResizeState({
      nodeId: null,
      handle: null,
      startX: 0,
      startY: 0,
      startWidth: 0,
      startHeight: 0,
    });
    setPanState({
      isPanning: false,
      startX: 0,
      startY: 0,
      startViewportX: 0,
      startViewportY: 0,
    });
    setCanvasSelection(null);
  };

  const handleWheel = (e: WheelEvent<HTMLDivElement>) => {
    if (!containerRef.current) return;
    e.preventDefault();
    e.stopPropagation();
    const rect = containerRef.current.getBoundingClientRect();
    const mx = e.clientX - rect.left;
    const my = e.clientY - rect.top;
    const delta = -e.deltaY;
    const zoomFactor = delta > 0 ? 1.1 : 1 / 1.1;

    setViewport((v) => {
      const newScale = Math.min(3, Math.max(0.2, v.scale * zoomFactor));
      const worldX = (mx - v.x) / v.scale;
      const worldY = (my - v.y) / v.scale;
      const newX = mx - worldX * newScale;
      const newY = my - worldY * newScale;
      return { x: newX, y: newY, scale: newScale };
    });
  };

  /** Mini-map 上的 slider 直接改變縮放（以畫布中心為基準） */
  const handleZoomSliderChange = (
    e: React.ChangeEvent<HTMLInputElement>
  ) => {
    if (!containerRef.current) {
      setViewport((v) => ({
        ...v,
        scale: Math.min(3, Math.max(0.2, Number(e.target.value) / 100)),
      }));
      return;
    }
    const target = Number(e.target.value);
    const newScale = Math.min(3, Math.max(0.2, target / 100));
    const rect = containerRef.current.getBoundingClientRect();
    const cx = rect.width / 2;
    const cy = rect.height / 2;
    setViewport((v) => {
      const worldX = (cx - v.x) / v.scale;
      const worldY = (cy - v.y) / v.scale;
      const newX = cx - worldX * newScale;
      const newY = cy - worldY * newScale;
      return { x: newX, y: newY, scale: newScale };
    });
  };

  /** Fit to screen */
  const fitToScreen = () => {
    if (!containerRef.current || !nodes.length) return;
    const rect = containerRef.current.getBoundingClientRect();

    let minX = Infinity;
    let minY = Infinity;
    let maxX = -Infinity;
    let maxY = -Infinity;

    nodes.forEach((node) => {
      minX = Math.min(minX, node.x);
      minY = Math.min(minY, node.y);
      maxX = Math.max(maxX, node.x + node.width);
      maxY = Math.max(maxY, node.y + node.height);
    });

    if (minX === Infinity || minY === Infinity) return;

    const boardWidth = maxX - minX;
    const boardHeight = maxY - minY;

    const margin = 50;
    const scaleX = (rect.width - margin * 2) / boardWidth;
    const scaleY = (rect.height - margin * 2) / boardHeight;
    const newScale = Math.max(0.2, Math.min(scaleX, scaleY));

    const worldCenterX = (minX + maxX) / 2;
    const worldCenterY = (minY + maxY) / 2;

    const viewportCenterX = rect.width / 2;
    const viewportCenterY = rect.height / 2;

    const newX = viewportCenterX - worldCenterX * newScale;
    const newY = viewportCenterY - worldCenterY * newScale;

    setViewport({ x: newX, y: newY, scale: newScale });
  };

  /** 拖曳模板產生節點 */
  const createNodeFromTemplate = (tpl: Node) => {
    recordHistory();
    const newNode: Node = {
      ...tpl,
      id: createId(),
      x: tpl.x + 20,
      y: tpl.y + 20,
    };
    setNodes((prev) => [...prev, newNode]);
  };

  const handleDrop = (e: React.DragEvent<HTMLDivElement>) => {
    e.preventDefault();
    if (!containerRef.current) return;
    const dataType = e.dataTransfer.getData("application/x-template-type");
    const tplId = e.dataTransfer.getData("application/x-template-id");
    const rect = containerRef.current.getBoundingClientRect();
    const worldX = (e.clientX - rect.left - viewport.x) / viewport.scale;
    const worldY = (e.clientY - rect.top - viewport.y) / viewport.scale;

    if (dataType === "template" && tplId) {
      const tpl = templateNodes.find((t) => t.id === tplId);
      if (tpl) {
        recordHistory();
        const newNode: Node = {
          ...tpl,
          id: createId(),
          x: worldX,
          y: worldY,
        };
        setNodes((prev) => [...prev, newNode]);
      }
    } else {
      const file = e.dataTransfer.files?.[0];
      if (file && file.type.startsWith("image/")) {
        const reader = new FileReader();
        reader.onload = (ev) => {
          const dataUrl = ev.target?.result as string;
          recordHistory();
          const newNode: ImageNode = {
            id: createId(),
            type: "image",
            x: worldX,
            y: worldY,
            width: 160,
            height: 160,
            imageDataUrl: dataUrl,
          };
          setNodes((prev) => [...prev, newNode]);
        };
        reader.readAsDataURL(file);
      }
    }
  };

  const handleDragOver = (e: React.DragEvent<HTMLDivElement>) => {
    e.preventDefault();
  };

  const handleResizeHandleMouseDown = (
    e: MouseEvent<HTMLDivElement>,
    node: Node,
    handle: ResizeHandle
  ) => {
    e.stopPropagation();
    if (!containerRef.current) return;
    const rect = containerRef.current.getBoundingClientRect();
    const startX = (e.clientX - rect.left - viewport.x) / viewport.scale;
    const startY = (e.clientY - rect.top - viewport.y) / viewport.scale;
    setResizeState({
      nodeId: node.id,
      handle,
      startX,
      startY,
      startWidth: node.width,
      startHeight: node.height,
    });
    isResizingRef.current = true;
    recordHistory();
  };

  const handleToolChange = (newTool: Tool) => {
    setTool(newTool);
  };

  const handleDoubleClickNode = (node: Node) => {
    setEditingId(node.id);
  };

  const handleNodeTextChange = (nodeId: string, newText: string) => {
    setNodes((prev) =>
      prev.map((node) => (node.id === nodeId ? { ...node, text: newText } : node))
    );
  };

  const handleNodeTableCellChange = (
    nodeId: string,
    rowIndex: number,
    colIndex: number,
    value: string
  ) => {
    setNodes((prev) =>
      prev.map((node) => {
        if (node.id !== nodeId || node.type !== "table") return node;
        const newCells = node.cells.map((row, r) =>
          row.map((cell, c) => (r === rowIndex && c === colIndex ? value : cell))
        );
        return { ...node, cells: newCells };
      })
    );

  const handleFileInputChange = async (
    e: React.ChangeEvent<HTMLInputElement>
  ) => {
    try {
      const file = e.target.files?.[0];
      if (!file) return;

      const name = file.name.toLowerCase();

      // 1) .lmwb / .json → 專案檔
      if (name.endsWith(".lmwb") || name.endsWith(".json")) {
        // 1-1) 若是 .lmwb，先嘗試當作 ZIP 專案檔處理
        if (name.endsWith(".lmwb")) {
          try {
            const buf = await file.arrayBuffer();
            const zip = await JSZip.loadAsync(buf);
            const projFile = zip.file("project.json");
            if (projFile) {
              const text = await projFile.async("text");
              const project = JSON.parse(text) as ProjectFileV2;
              if (project.version === 2 && project.pages?.length) {
                recordHistory();
                importProject(project);
                return;
              }
            }
          } catch (zipErr) {
            console.warn(
              "載入 .lmwb ZIP 專案失敗，改用純文字 JSON 解析。",
              zipErr
            );
            // 繼續往下用純文字 JSON parse
          }
        }

        // 1-2) 純文字 JSON 專案檔
        const text = await file.text();
        try {
          const project = JSON.parse(text) as ProjectFileV2;
          if (project.version === 2 && project.pages?.length) {
            recordHistory();
            importProject(project);
          } else {
            alert("不支援的檔案版本（只支援 version = 2）。");
          }
        } catch (err) {
          console.error("解析專案檔案失敗", err);
          alert("解析專案檔案失敗，請確認格式是否正確。");
        } finally {
          e.target.value = "";
        }
        return;
      }

      // 2) .zip → 文字大綱匯入成節點（保留原本功能）
      if (name.endsWith(".zip")) {
        const arrayBuffer = await file.arrayBuffer();
        const zip = await JSZip.loadAsync(arrayBuffer);
        const txtEntry = Object.values(zip.files).find((f) =>
          f.name.toLowerCase().endsWith(".txt")
        );
        if (!txtEntry) {
          alert("ZIP 檔內未找到 .txt 策略匯出檔。");
          return;
        }
        const txtContent = await txtEntry.async("text");
        const lines = txtContent.split(/\r?\n/);
        const newNodes: Node[] = [];
        let y = 80;
        for (const line of lines) {
          const trimmed = line.trim();
          if (!trimmed) continue;

          const isL1 = /^\d+[\.\)]/.test(trimmed);
          const isL2 = /^\s{2,}\d+[\.\)]/.test(trimmed);
          const isL3 = /^\s{4,}\d+[\.\)]/.test(trimmed);

          let level: StrategyLevel = "Info";
          let x = 200;
          let width = 260;
          if (isL1) {
            level = "L1";
            x = 80;
            width = 280;
          } else if (isL2) {
            level = "L2";
            x = 380;
            width = 260;
          } else if (isL3) {
            level = "L3";
            x = 680;
            width = 260;
          }

          const text = trimmed.replace(/^\s*\d+[\.\)]\s*/, "");
          newNodes.push({
            id: createId(),
            type: "note",
            x,
            y,
            width,
            height: 60,
            text,
            bgColor:
              level === "L1"
                ? "#fee2e2"
                : level === "L2"
                ? "#e0f2fe"
                : level === "L3"
                ? "#ecfeff"
                : "#e5e7eb",
            fontSize: 13,
            level,
          });
          y += 80;
        }
        recordHistory();
        setNodes((prev) => [...prev, ...newNodes]);
        return;
      }

      alert("只支援 .lmwb / .json / .zip 檔案匯入。");
    } catch (err) {
      console.error("匯入檔案時發生錯誤", err);
      alert("匯入檔案失敗，請稍後再試。");
    } finally {
      e.target.value = "";
    }

                          formatPainterActive
                            ? "bg-amber-500 text-white"
                            : "bg-slate-100 text-slate-700 hover:bg-slate-200"
                        }`}
                        onClick={handleFormatPainterToggle}
                      >
                        {formatPainterActive ? "停用格式刷" : "啟動格式刷"}
                      </button>
                    </div>
                    <div className="text-[11px] text-slate-500">
                      啟動後，選取其它節點並按下「套用格式」，可將目前節點的顏色
                      / Level / Tag 等格式套用到多個節點。
                    </div>
                    <button
                      className="mt-1 w-full rounded bg-slate-100 py-1 text-[11px] text-slate-700 hover:bg-slate-200"
                      onClick={handleApplyFormatPainter}
                    >
                      對目前選取節點套用格式
                    </button>
                  </div>

                  {selectedIds.length >= 2 && (
                    <div className="border-t border-slate-200 pt-2 text-[11px] text-slate-500">
                      多選排列工具可在上方 Arrange 工具列中使用。
                    </div>
                  )}
                </div>
              )}
            </div>
          </aside>
        )}

        {!isRightPanelOpen && (
          <button
            className="absolute right-1 top-[60px] z-20 rounded border border-slate-300 bg-slate-50 px-1 py-0.5 text-[11px] text-slate-500 hover:bg-slate-100"
            onClick={toggleRightPanel}
          >
            屬性
          </button>
        )}
      </div>
    </div>
  );
}

/** 節點四周的 resize 小圓點 */
function ResizeHandleBox({
  node,
  onResizeHandleMouseDown,
}: {
  node: Node;
  onResizeHandleMouseDown: (
    e: MouseEvent<HTMLDivElement>,
    node: Node,
    handle: ResizeHandle
  ) => void;
}) {
  const handleBase =
    "absolute h-2 w-2 bg-white border border-blue-500 rounded-full";

  return (
    <>
      <div
        className={handleBase}
        style={{ left: -4, top: -4, cursor: "nwse-resize" }}
        onMouseDown={(e) => onResizeHandleMouseDown(e, node, "top-left")}
      />
      <div
        className={handleBase}
        style={{
          left: "50%",
          top: -4,
          transform: "translateX(-50%)",
          cursor: "ns-resize",
        }}
        onMouseDown={(e) => onResizeHandleMouseDown(e, node, "top")}
      />
      <div
        className={handleBase}
        style={{ right: -4, top: -4, cursor: "nesw-resize" }}
        onMouseDown={(e) => onResizeHandleMouseDown(e, node, "top-right")}
      />
      <div
        className={handleBase}
        style={{
          right: -4,
          top: "50%",
          transform: "translateY(-50%)",
          cursor: "ew-resize",
        }}
        onMouseDown={(e) => onResizeHandleMouseDown(e, node, "right")}
      />
      <div
        className={handleBase}
        style={{ right: -4, bottom: -4, cursor: "nwse-resize" }}
        onMouseDown={(e) =>
          onResizeHandleMouseDown(e, node, "bottom-right")
        }
      />
      <div
        className={handleBase}
        style={{
          left: "50%",
          bottom: -4,
          transform: "translateX(-50%)",
          cursor: "ns-resize",
        }}
        onMouseDown={(e) => onResizeHandleMouseDown(e, node, "bottom")}
      />
      <div
        className={handleBase}
        style={{ left: -4, bottom: -4, cursor: "nesw-resize" }}
        onMouseDown={(e) =>
          onResizeHandleMouseDown(e, node, "bottom-left")
        }
      />
      <div
        className={handleBase}
        style={{
          left: -4,
          top: "50%",
          transform: "translateY(-50%)",
          cursor: "ew-resize",
        }}
        onMouseDown={(e) => onResizeHandleMouseDown(e, node, "left")}
      />
    </>
  );
}
